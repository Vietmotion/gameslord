Simulate getting a score in your game:

📝 How to integrate in your games:

// 1. When game loads, display player's high score:
const playerHighScore = await window.parent.GameScores.getPlayerHighScore('smot');
console.log('Your best:', playerHighScore);

// 2. When game loads, display leaderboard:
const leaderboard = await window.parent.GameScores.getLeaderboard('smot', 5);
console.log('Top 5:', leaderboard);

// 3. When game ends, save the score:
const finalScore = 1234; // Your game's score
const result = await window.parent.GameScores.saveScore('smot', finalScore);
if (result.success) {
    console.log('New high score!');
}

// 4. Check if user is logged in:
if (window.parent.GameScores.isLoggedIn()) {
    // Show score features
} else {
    // Show "login to save scores" message
}
